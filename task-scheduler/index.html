Build a Concurrency Controlled Task Scheduler

Create a task scheduler that limits concurrent async task execution.

Factory:

createScheduler({ concurrency })

Returned methods:
  • add(task, priority)
  • cancel(taskId)
Task:
  • A Function returning a Promise
Requirements:
  1. Concurrency Control
    • Only N tasks run at the same time.
    • Remaining tasks are queued.
  2. Priority Support
    • Higher priority tasks should execute first.
    • Must not break running tasks.
  3. Cancellation
    • Tasks in queue can be cancelled.
    • Running tasks may support AbortController optionally.
  4. Completion Handling
    • Scheduler must automatically start next queued task after one completes.
  5. Error Handling
    • Failed tasks should not block queue.
    • Errors should be properly surfaced.
Constraints:
  • No external libraries.
  • Avoid race conditions.
  • Design clean queue abstraction.
Evaluation Criteria:
  • Correct concurrency limiting
  • Proper queue management
  • Clean async control flow
  • Edge case handling
  • Code clarity

function createScheduler({ concurrency }) {
	const queue = [];
	let running = 0;
	let idCounter = 1;

	function updateUI() {
		document.getElementById("active-count").innerText = running;
		document.getElementById("queue-count").innerText = queue.length;
	}

	function run(taskItem) {
		running++;
		updateUI();

		const controller = new AbortController();

		taskItem
			.task(controller.signal)
			.then(() => {
				log(`Task ${taskItem.id} completed`);
			})
			.catch((err) => {
				log(`Task ${taskItem.id} stopped: ${err.message}`);
			})
			.finally(() => {
				running--;

				if (queue.length) {
					let highestIndex = 0;

					for (let i = 1; i < queue.length; i++) {
						if (queue[i].priority > queue[highestIndex].priority) {
							highestIndex = i;
						}
					}

					const next = queue.splice(highestIndex, 1)[0];
					run(next);
				}

				updateUI();
			});
	}

	return {
		add(task, priority) {
			const id = idCounter++;

			const item = { id, task, priority };

			if (running < concurrency) {
				setTimeout(() => run(item), 0);
			} else {
				queue.push(item);
				updateUI();
			}

			return id;
		},

		cancel(id) {
			const idx = queue.findIndex((t) => t.id === id);

			if (idx !== -1) {
				queue.splice(idx, 1);
				updateUI();
				return true;
			}
			return false;
		},
	};
}

const scheduler = createScheduler({ concurrency: 1 });
let lastTaskId = null;

function log(msg) {
	const logEl = document.getElementById("log");
	logEl.innerHTML += `
> ${msg}
`;
	logEl.scrollTop = logEl.scrollHeight;
}

function addTask(priority, duration) {
	let id;

	id = scheduler.add((signal) => {
		log(`START: Task #${id} (P:${priority})`);

		return new Promise((resolve, reject) => {
		const timer = setTimeout(() => {
			log(`END: Task #${id}`);
			resolve();
		}, duration);

		signal.onabort = () => {
			clearTimeout(timer);
			reject(new Error("Cancelled"));
		};
		});
	}, priority);

	lastTaskId = id;
	log(`Queued Task #${id} (Priority ${priority})`);
}

function cancelLast() {
	if (lastTaskId) {
		if (scheduler.cancel(lastTaskId)) {
			log(`Cancelled Task #${lastTaskId}`);
		} else {
			log(`Task #${lastTaskId} already running or finished`);
		}
	} else {
		log("No task to cancel");
	}
}

function testPriorityCase() {
	addTask(1, 2000);
	addTask(1, 3000);
	addTask(10, 3000);
